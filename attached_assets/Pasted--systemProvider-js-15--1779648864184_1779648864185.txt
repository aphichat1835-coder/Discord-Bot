BUG #1 — deleteSession() ทำลาย Client ที่ยังใช้อยู่ร่วมกัน

ไฟล์: sessionManager.js Region 7

// sessionManager.js บรรทัด 312
async function deleteSession(sessionId) {
    ...
    if (session.client) { try { session.client.destroy(); } catch {} } // ← ปัญหา
}


ปัญหา: session.client เป็น reference ไปที่ object เดียวกับใน clientPool ใน voiceWorker.js ถ้ามี 2 Sessions ที่ใช้ Token เดียวกัน (เช่น user เดียวใน 2 เซิร์ฟ) — การลบ Session แรกจะ destroy() selfbot client ที่ Session ที่สองยังใช้อยู่ด้วย ทำให้ Session ที่สองหลุดเงียบๆ โดยไม่มีการแจ้งเตือน

แก้ไข: ลบบรรทัด session.client.destroy() ออกจาก deleteSession() ปล่อยให้ voiceWorker.stopSession() จัดการ client lifecycle เองผ่าน clientPool

BUG #4 — botReadyAt ถูก Freeze เป็น null ตั้งแต่ Boot

ไฟล์: index.js บรรทัด 171, index/server.js บรรทัด 136

// index.js — ส่งเป็น getter
registerRoutes({
    get botReadyAt() { return system.botReadyAt; }, // ← getter
});

// index/server.js — destructure แล้ว evaluate ทันที
function registerRoutes({ ..., botReadyAt, ... }) {
    // botReadyAt ถูก evaluate ตอน registerRoutes() ถูกเรียก
    // ซึ่งเป็นช่วง boot ก่อน client.ready → ค่า = null ตลอดไป
    const botOnlineSec = (typeof botReadyAt === 'function' ? botReadyAt() : botReadyAt)
    // typeof null === 'object' ไม่ใช่ 'function' → botOnlineSec = null ทุกครั้ง
}


ปัญหา: /api/status คืนค่า botOnlineSec: null ตลอด แม้บอทออนแล้ว

แก้ไข: เปลี่ยนจาก getter เป็น function:

// index.js
registerRoutes({
    getBotReadyAt: () => system.botReadyAt,
});
// server.js
const botOnlineSec = getBotReadyAt()
    ? Math.floor((Date.now() - getBotReadyAt()) / 1000) : null;

BUG #5 — DB Query ทุกข้อความที่ส่ง (Performance Killer)

ไฟล์: index/events.js บรรทัด 26

client.on("messageCreate", async (message) => {
    // ⚠️ ทำ MongoDB query ทุกครั้งที่มีใครส่งข้อความ
    const antiRaidEnabled = await sessionManager.getSetting('antiRaidEnabled', true);


ปัญหา: ถ้ามี 10 เซิร์ฟเวอร์ แต่ละเซิร์ฟส่งข้อความ 100 ครั้ง/นาที = DB query 1,000 ครั้ง/นาที เฉพาะฟีเจอร์นี้อย่างเดียว

แก้ไข: Cache ค่าไว้ใน memory แล้วอัปเดตจาก DB ทุก 60 วินาที:

let _antiRaidCached = true;
let _antiRaidCacheTs = 0;
async function getAntiRaidEnabled() {
    if (Date.now() - _antiRaidCacheTs > 60000) {
        _antiRaidCached = await sessionManager.getSetting('antiRaidEnabled', true);
        _antiRaidCacheTs = Date.now();
    }
    return _antiRaidCached;
}
