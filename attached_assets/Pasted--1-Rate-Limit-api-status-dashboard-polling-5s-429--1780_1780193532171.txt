🟡 จุดที 2 — Input Sanitization สำหรับ /say และ /announce

แก้ใน discord/commands/utility.js เพิ่มฟังก์ชัน sanitize ก่อน handleSay:

// ══════ เพิ่ม helper ตรงนี้ ก่อน handle() ══════

const BLOCKED_PATTERNS = [
    /discord\.gg\/\S+/gi,           // invite links
    /https?:\/\/\S+\.(exe|bat|cmd|sh|ps1)/gi, // ลิงก์ไฟล์อันตราย
];

function sanitizeMessage(msg) {
    if (!msg || typeof msg !== 'string') return '';

    // จำกัดความยาว
    let clean = msg.slice(0, 1000);

    // ลบ @everyone และ @here ออก (ยกเว้น admin จดการใน permission แล้ว)
    clean = clean.replace(/@everyone/g, '@\u200beveryone');
    clean = clean.replace(/@here/g,     '@\u200bhere');

    // บล็อก pattern อันตราย
    for (const pattern of BLOCKED_PATTERNS) {
        clean = clean.replace(pattern, '[ลิงก์ถูกบล็อก]');
    }

    return clean.trim();
}
