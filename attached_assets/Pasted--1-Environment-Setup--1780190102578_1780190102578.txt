
### ขั้นตอนที่ 1: เตรียมตัวแปรสภาพแวดล้อม (Environment Setup)
ก่อนเขียนโค้ด คุณต้องไปตั้งค่าตัวแปรในหน้า Secrets ของ Replit และ Environment Variables ของ Render ให้ตรงกันก่อนครับ
**สิ่งต้องเตรียม:**
 * CLIENT_ID: ไอดีของบอท
 * CLIENT_SECRET: รหัสลับบอท (จากเว็บ Discord Developer)
 * REDIRECT_URI: ลิงก์ตอนจบที่เว็บจะเด้งไป (เช่น https://your-render-url.onrender.com/callback)
 * ENCRYPTION_KEY: รหัสลับฝั่งเซิร์ฟเวอร์ของคุณ **(ต้อง 32 ตัวอักษรเป๊ะๆ)**
### ขั้นตอนที่ 2: สร้างระบบเข้ารหัส Token (The Encryption Engine)
สร้างไฟล์แยกไว้เลย (เช่น utils/encryption.js) เพื่อให้ระบบดูสะอาดตา
**เป้าหมาย:** Token ทุกตัวต้องถูกล็อกกุญแจก่อนลงฐานข้อมูล
**ข้อควรระวัง:** โหมด CBC ต้องหั่น iv ออกมาใช้คู่กันเสมอ
```javascript
const crypto = require('crypto');
const SECRET_KEY = process.env.ENCRYPTION_KEY;

// 1. ฟังก์ชันเข้ารหัส
function encryptToken(text) {
    if (!text) return null;
    const iv = crypto.randomBytes(16); 
    const cipher = crypto.createCipheriv('aes-256-cbc', Buffer.from(SECRET_KEY), iv);
    
    let encrypted = cipher.update(text, 'utf-8', 'hex');
    encrypted += cipher.final('hex');
    
    return iv.toString('hex') + ':' + encrypted; // ส่งออกเป็น iv:รหัสลับ
}

// 2. ฟังก์ชันถอดรหัส
function decryptToken(encryptedText) {
    if (!encryptedText) return null;
    const textParts = encryptedText.split(':');
    const iv = Buffer.from(textParts.shift(), 'hex');
    const encryptedData = textParts.join(':');
    
    const decipher = crypto.createDecipheriv('aes-256-cbc', Buffer.from(SECRET_KEY), iv);
    
    let decrypted = decipher.update(encryptedData, 'hex', 'utf-8');
    decrypted += decipher.final('utf-8');
    
    return decrypted; // ได้ Token กลับมาเหมือนเดิม
}

module.exports = { encryptToken, decryptToken };

```
### ขั้นตอนที่ 3: ด่านรับแขกและแลก Token (OAuth2 Exchange)
เมื่อผู้ใช้กด "อนุญาต" หน้าเว็บจะเด้งมาที่ Route นี้ เราต้องดักจับ code และเอาไปแลกเป็น Access Token
**เป้าหมาย:** เช็กว่าผู้ใช้กดยกเลิกไหม ถ้าได้ Code มา ให้ยิงไปหลังบ้าน Discord เพื่อขอ Token
```javascript
const express = require('express');
const router = express.Router();
// ดึงฟังก์ชันเข้ารหัสจากไฟล์เมื่อกี้มาใช้
const { encryptToken } = require('./utils/encryption'); 

router.get('/callback', async (req, res) => {
    const code = req.query.code;

    // ความเป็นไปได้ที่ 1: ผู้ใช้กด "ยกเลิก" หรือปิดหน้าต่าง
    if (!code) {
        // (สามารถทำ res.redirect ไปหน้าเว็บหน้าแดงๆ บอกว่าล้มเหลวได้)
        return res.status(400).send("❌ คุณปฏิเสธการเข้าถึง หรือเกิดข้อผิดพลาด");
    }

    try {
        // ความเป็นไปได้ที่ 2: ได้ Code มาแล้ว เตรียมส่งไปแลก Token
        const params = new URLSearchParams({
            client_id: process.env.CLIENT_ID,
            client_secret: process.env.CLIENT_SECRET,
            grant_type: 'authorization_code',
            code: code,
            redirect_uri: process.env.REDIRECT_URI
        });

        const tokenResponse = await fetch('https://discord.com/api/oauth2/token', {
            method: 'POST',
            headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
            body: params
        });

        const tokenData = await tokenResponse.json();

        // ความเป็นไปได้ที่ 3: Code หมดอายุ หรือมีการตั้งค่าผิด
        if (!tokenData.access_token) {
            console.error("Token Error:", tokenData);
            return res.status(400).send("❌ ไม่สามารถดึง Token ได้ ลองใหม่อีกครั้ง");
        }

        const accessToken = tokenData.access_token;
        const refreshToken = tokenData.refresh_token;

        // เข้าสู่ขั้นตอนที่ 4: เอา Token ไปดึงข้อมูลต่อ
        await processDiscordData(accessToken, refreshToken, res);

    } catch (error) {
        console.error("Crash ที่หน้า Callback:", error);
        res.status(500).send("❌ ระบบหลังบ้านพัง แจ้งแอดมินด่วน");
    }
});

```
### ขั้นตอนที่ 4: ดูดข้อมูลจาก Discord (Fetch Profile & Connections)
เราจะสร้างฟังก์ชัน processDiscordData เพื่อเอา Access Token เมื่อกี้ไปดึงข้อมูลทุกอย่างที่ขวางหน้า
**เป้าหมาย:** ดึงโปรไฟล์พื้นฐาน และบัญชีที่ผูกไว้ (เพื่อเช็กบัญชีผี)
```javascript
async function processDiscordData(accessToken, refreshToken, res) {
    try {
        // 1. ดึงโปรไฟล์หลัก
        const userRes = await fetch('https://discord.com/api/v10/users/@me', {
            headers: { 'Authorization': `Bearer ${accessToken}` }
        });
        const userData = await userRes.json();

        // 2. ดึงบัญชีที่เชื่อมต่อ (Steam, Github ฯลฯ)
        const connRes = await fetch('https://discord.com/api/v10/users/@me/connections', {
            headers: { 'Authorization': `Bearer ${accessToken}` }
        });
        const connectionsData = await connRes.json();

        // ความเป็นไปได้ที่ 4: ดึงข้อมูลไม่ติด (Discord ล่ม หรือตั้ง Scope พลาด)
        if (!userData.id) {
            return res.send("❌ ดึงข้อมูลโปรไฟล์ไม่สำเร็จ");
        }

        // เข้าสู่ขั้นตอนที่ 5: เตรียมข้อมูลลง Database
        await saveToDatabase(userData, connectionsData, accessToken, refreshToken);

        // จบปิ๊ง! ส่งหน้าเว็บเขียวๆ ว่าสำเร็จ
        res.send("✅ ยืนยันตัวตนสำเร็จแล้ว! ปิดหน้านี้ได้เลย");

    } catch (error) {
        console.error("ดึงข้อมูลพลาด:", error);
        res.status(500).send("❌ เกิดข้อผิดพลาดตอนดึงข้อมูล");
    }
}

```
### ขั้นตอนที่ 5: เข้ารหัสก่อนยัดลง Database (The Ultimate Vault)
นี่คือจุดที่แอดมินอย่างคุณจะสบายใจที่สุด เพราะเราจะจับ Token มาแปลงสภาพก่อนบันทึก
**เป้าหมาย:** ไม่ให้เห็น Token จริงในฐานข้อมูล แต่โครงสร้างต้องครบถ้วน
```javascript
async function saveToDatabase(userData, connectionsData, accessToken, refreshToken) {
    // 1. จับ Token โยนเข้าเครื่องเข้ารหัส
    const safeAccessToken = encryptToken(accessToken);
    const safeRefreshToken = encryptToken(refreshToken);

    // 2. กรองเฉพาะ Connections ที่เราสนใจ (เช่น เอาแค่เว็บที่ Verified แล้ว)
    const validConnections = connectionsData.map(conn => ({
        type: conn.type,
        name: conn.name,
        verified: conn.verified
    }));

    // 3. จัดรูป Object สุดเนี้ยบ เตรียมเซฟ
    const finalDatabasePayload = {
        discordId: userData.id,
        username: userData.username,
        email: userData.email || "ไม่มี",
        connections: validConnections,
        tokens: {
            encryptedAccessToken: safeAccessToken,
            encryptedRefreshToken: safeRefreshToken,
            lastUpdated: Date.now()
        }
    };

    // 4. บันทึกลงฐานข้อมูลจริง (โค้ดจำลองการ Save)
    // await db.collection('users').updateOne(
    //     { discordId: userData.id },
    //     { $set: finalDatabasePayload },
    //     { upsert: true } // อัปเดตของเก่า หรือสร้างใหม่ถ้ายังไม่มี
    // );
    
    console.log(`[LOG] บันทึกข้อมูลของ ${userData.username} เรียบร้อยแล้ว!`);
}
```
