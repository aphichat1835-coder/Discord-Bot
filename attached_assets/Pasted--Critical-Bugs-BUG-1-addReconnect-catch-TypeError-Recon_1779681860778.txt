🔴 Critical Bugs ที่ยังเหลือ

BUG #1 — addReconnect().catch() TypeError → ทำลายระบบ Reconnect ทั้งหมด
ไฟล์: voiceWorker.js — connectToVoice() Disconnected handler
js// voiceWorker.js
connection.on(VoiceConnectionStatus.Disconnected, async () => {
    reconnectAttempts++;
    sessionManager.addReconnect(sessionId).catch(() => {}); // ← 💣 CRASH
    console.log(`[WORKER] ⚠️ Voice dropped...`); // ← ไม่ถูกเรียกเลย
    // ... reconnect logic ทั้งหมด ← ไม่ถูกเรียกเลย
});
ปัญหา: addReconnect() คืนค่าเป็น number (ไม่ใช่ Promise)
js// sessionManager.js
function addReconnect(sessionId) {
    // ...
    return history.length; // ← number!
}
JavaScript execution:
sessionManager.addReconnect(sessionId)  →  1  (number)
(1).catch                               →  undefined (no .catch on numbers)
undefined(() => {})                     →  TypeError: undefined is not a function ← THROWS
เนื่องจากอยู่ใน async function — TypeError นี้ทำให้ function reject ทันที โค้ดที่อยู่หลัง .catch() ไม่ถูก execute เลย รวมถึง exponential backoff, passive reconnect, urgent recovery — ระบบ reconnect ทั้งหมดหยุดทำงาน ทุกครั้งที่ voice หลุด
แก้ไข:
js// ลบ .catch() ออก — addReconnect เป็น sync ไม่ต้องการ .catch
sessionManager.addReconnect(sessionId);

jsconst prevHistory = (sayUsageTracking.get(userId) || []).filter(t => now - t < 60000);

แก้ไข: Cache ต่อ guildId ใน Memory Map แล้ว invalidate เมื่อ /setup-log รัน:
jsconst auditChannelCache = new Map(); // guildId → { map, expiry }

async function getAuditChannel(guild, sessionManager, type) {
    const now = Date.now();
    let cached = auditChannelCache.get(guild.id);
    if (!cached || now > cached.expiry) {
        const map = await sessionManager.getLogChannelMap(guild.id);
        cached = { map, expiry: now + 300000 }; // cache 5 นาที
        auditChannelCache.set(guild.id, cached);
    }
    const channelId = cached.map?.[`${type}ChannelId`];
    return channelId ? guild.channels.cache.get(channelId) || null : null;
}
